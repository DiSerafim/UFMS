Tipos de persistência de dados.

  • Shared Preferences/Data Store: armazenar dados primitivos em pares chave-valor. Semelhante a um Hashmap;
  • Internal Storage: armazenar dados privados na memória do dispositivo (com persistência de objetos);
  • External Storage: armazenar dados privados/públicos em uma memória externa (compartilhada);
  • SQLite Databases: armazenar dados estruturados em um banco de dados privado.

Banco de Dados SQLite.

  • O SQLite é uma biblioteca que implementa um BD com as seguintes características:
    • Transacional: permite executar um conjunto de operações e só efetivá-las quando todas foram bem sucedidas;
    • Autocontido: mínima dependência do SO;
    • Trabalha localmente: não fica hospedado em um servidor;
    • Não precisa ser instalado ou configurado;
    • Fica em um arquivo no armazenamento interno.
  • Permite ainda:
    • Criar tabelas com PK e FK;
    • Views;
    • Índices;
    • Triggers.
  • Tipos de dados suportados:
    • Null, Integer, Real, Text, Blob.
  • Para trabalhar com SQLite é preciso criar uma classe que herde SQLiteOpenHelper.
  • Opcional:
    • Criar classes abstratas e estáticas para representar cada tabela do BD;
    • Campos das tabelas do BD tornam-se atributos das classes.
  • Métodos:
    • onCreate(): para criar um banco;
    • onUpgrade(): para atualizar um banco;
    • execSQL(): executa uma instrução SQL que não seja um Select;
    • query(): realiza uma consulta no banco.
  • Objeto SQLiteDatabase:
    • Permite executar os comandos SQL.

MainActivity.java 
            package com.example.crud_android;

            import android.content.ContentValues;
            import android.database.Cursor;
            import android.database.sqlite.SQLiteDatabase;
            import android.os.Bundle;
            import android.view.View;
            import android.widget.EditText;
            import android.widget.TextView;

            import androidx.activity.EdgeToEdge;
            import androidx.appcompat.app.AppCompatActivity;
            import androidx.core.graphics.Insets;
            import androidx.core.view.ViewCompat;
            import androidx.core.view.WindowInsetsCompat;

            public class MainActivity extends AppCompatActivity {

                private EditText etNome, etRA;
                private TextView tvResultado;
                private SQLiteDatabase banco;

                @Override
                protected void onCreate(Bundle savedInstanceState) {
                    super.onCreate(savedInstanceState);
                    EdgeToEdge.enable(this);
                    setContentView(R.layout.activity_main);
                    ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main), (v, insets) -> {
                        Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
                        v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom);
                        return insets;
                    });

                    // Associa aos componentes da tela
                    etNome = (EditText) findViewById(R.id.etNome);
                    etRA = (EditText) findViewById(R.id.etRA);
                    tvResultado = (TextView) findViewById(R.id.tvResultado);

                    // Gerenciador
                    GerenciadoraBD gerenciadoraBD = new GerenciadoraBD(this);

                    // Liga o gerenciador ao banco
                    banco = gerenciadoraBD.getWritableDatabase();
                }

                // Assinaturas dos métodos Inseri, Apagar, Atualizar e Buscar
                public void inserir(View v) {
                    // Tratamento de campos vazios
                    if (etNome.getText().toString().isEmpty() || etRA.getText().toString().isEmpty()) {
                        tvResultado.setText("Preencha todos os campos!");
                        return;
                    }
                    try {
                        ContentValues valores = new ContentValues();
                        // Tabela em que valores serão inseridos
                        valores.put(GerenciadoraBD.TBAluno.NOME, etNome.getText().toString());
                        valores.put(GerenciadoraBD.TBAluno.RA, Integer.valueOf(etRA.getText().toString()));

                        // Isere os valores no banco
                        banco.insert(GerenciadoraBD.TBAluno.TABELA, null, valores);
                        tvResultado.setText("Dados inseridos!");
                    } catch (NumberFormatException e) {
                        tvResultado.setText("RA deve ser um número!");
                    }
                }

                public void apagar(View v) {
                    String calusulaWhere = "RA = ?";
                    String[] argumentoWhere = {etRA.getText().toString()};
                    banco.delete(GerenciadoraBD.TBAluno.TABELA, calusulaWhere, argumentoWhere);
                }

                public void atualizar(View v) {
                    ContentValues valores = new ContentValues();
                    valores.put(GerenciadoraBD.TBAluno.NOME, etNome.getText().toString());

                    String clausulaWhere = "RA = ?";
                    String[] argumentoWhere = {etRA.getText().toString()};

                    banco.update(GerenciadoraBD.TBAluno.TABELA, valores, clausulaWhere, argumentoWhere);
                }

                public void buscarTudo(View v) {
                    String[] colunas = {GerenciadoraBD.TBAluno.NOME, GerenciadoraBD.TBAluno.RA};
                    String selecao = "";
                    String[] argumentos = null;
                    String groupBy = null;
                    String having = null;
                    String orderBy = GerenciadoraBD.TBAluno._ID + " ASC"; // ou DESC

                    Cursor cursor;

                    cursor = banco.query(GerenciadoraBD.TBAluno.TABELA, colunas, selecao, argumentos, groupBy, having, orderBy);

                    String nome;
                    int ra;

                    while (cursor.moveToNext()) {
                        nome = cursor.getString(cursor.getColumnIndex(GerenciadoraBD.TBAluno.NOME));
                        ra = cursor.getInt(cursor.getColumnIndex(GerenciadoraBD.TBAluno.RA));

                        tvResultado.setText(tvResultado.getText() + "Nome: " + nome + " / " + "RA: " + ra + "\n");
                    }
                }
            }
GerenciadoraDB.java 
            package com.example.crud_android;

            import android.content.Context;
            import android.database.sqlite.SQLiteDatabase;
            import android.database.sqlite.SQLiteOpenHelper;
            import android.provider.BaseColumns;

            public class GerenciadoraBD extends SQLiteOpenHelper {

                // Cria as constants com 'static final'
                public static final String NOME_BANCO = "primeiro_bd";
                public  static final int VERSAO = 2;

                // Classes abstratas
                public static abstract class TBAluno implements BaseColumns {
                    // Atributos da classe (nome dos campos)
                    public static final String TABELA = "Aluno";
                    public static final String NOME = "Nome";
                    public static final String RA = "RA";

                }

                // Implementa o construtor da classe
                public GerenciadoraBD(Context context) {
                    super(context, NOME_BANCO, null, VERSAO);
                }
                @Override
                public void onCreate(SQLiteDatabase sqLiteDatabase) {
                    // SQL de construção do DB
                    StringBuilder builder = new StringBuilder();
                    String sql = builder.append("CREATE TABLE " + TBAluno.TABELA)
                            .append("(")
                            .append(TBAluno._ID + " INTEGER PRIMARY KEY, ")
                            .append(TBAluno.NOME + " TEXT NOT NULL, ")
                            .append(TBAluno.RA + " INTEGER NOT NULL ")
                            .append(")").toString();
                    sqLiteDatabase.execSQL(sql);
                }

                @Override
                public void onUpgrade(SQLiteDatabase sqLiteDatabase, int i, int i1) {

                }
            }
